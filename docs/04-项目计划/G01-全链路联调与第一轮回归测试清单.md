```sql

```

基础核验：

```sql
SELECT DATABASE();

SELECT COUNT(*) AS workorder_table_count
FROM information_schema.tables
WHERE table_schema = 'workorder' AND table_name LIKE 'wo\_%';

SELECT role_key, COUNT(*) AS role_count
FROM sys_role
WHERE role_key IN ('workorder_reporter', 'workorder_engineer', 'workorder_dispatcher')
  AND del_flag = '0'
GROUP BY role_key;

SELECT user_name, phonenumber, status, del_flag
FROM sys_user
WHERE user_name IN ('wo_reporter', 'wo_engineer', 'wo_dispatcher')
ORDER BY user_name;

SELECT job_name, job_group, status
FROM sys_job
WHERE job_group = 'WORKORDER'
ORDER BY job_id;
```
